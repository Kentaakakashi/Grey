<?php
require_once __DIR__ . '/bootstrap.php';

require_method('POST');

$data = read_json_body();
$setupKey = (string)($data['setup_key'] ?? '');

if (!hash_equals(SETUP_KEY, $setupKey)) {
    json_response(['success' => false, 'message' => 'Invalid setup key'], 403);
}

$username = trim((string)($data['username'] ?? 'admin'));
$password = (string)($data['password'] ?? '');

if ($username === '' || $password === '') {
    json_response(['success' => false, 'message' => 'username and password are required'], 400);
}

if (strlen($username) > 100 || strlen($password) < 8 || strlen($password) > 255) {
    json_response(['success' => false, 'message' => 'Invalid username or password length'], 400);
}

$conn = get_db_connection();
ensure_admin_users_table($conn);
ensure_app_state_table($conn);

$passwordHash = password_hash($password, PASSWORD_DEFAULT);

$stmt = $conn->prepare('INSERT INTO admin_users (username, password_hash) VALUES (?, ?) ON DUPLICATE KEY UPDATE password_hash = VALUES(password_hash)');

if (!$stmt) {
    error_log('setup admin prepare failed: ' . $conn->error);
    $conn->close();
    json_response(['success' => false, 'message' => 'Setup failed'], 500);
}

$stmt->bind_param('ss', $username, $passwordHash);

if (!$stmt->execute()) {
    error_log('setup admin execute failed: ' . $stmt->error);
    $stmt->close();
    $conn->close();
    json_response(['success' => false, 'message' => 'Setup failed'], 500);
}

$stmt->close();

$initialData = $data['initial_data'] ?? null;
if (is_array($initialData)) {
    $jsonData = json_encode($initialData, JSON_UNESCAPED_UNICODE | JSON_UNESCAPED_SLASHES);
    if ($jsonData !== false) {
        $seedStmt = $conn->prepare('INSERT INTO app_state (id, json_data) VALUES (1, ?) ON DUPLICATE KEY UPDATE json_data = VALUES(json_data)');
        if ($seedStmt) {
            $seedStmt->bind_param('s', $jsonData);
            if (!$seedStmt->execute()) {
                error_log('setup initial_data execute failed: ' . $seedStmt->error);
            }
            $seedStmt->close();
        }
    }
}

$conn->close();

json_response([
    'success' => true,
    'message' => 'Setup complete',
    'created_admin' => $username,
]);
